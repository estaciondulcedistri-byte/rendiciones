function nkPress(key) {

else npValor += key;

function actualizarNumpadDisplay() {
const v = parseInt(npValor) || 0; const nc = npMontoOrig - v;
const disp = document.getElementById('np-display'); const ncEl = document.getElementById('np-nc');
if (!npValor) { disp.innerHTML = '<span class="hint">Ingresá el monto cobrado</span>'; disp.style.color = ''; ncEl.textContent = ''; ncEl.style.color = ''; }
else if (v > npMontoOrig) { disp.textContent = fmt(v); disp.style.color = '#c0392b'; ncEl.textContent = 'Supera el original (' + fmt(npMontoOrig) + ')'; ncEl.style.color = '#c0392b'; }
else if (nc > 0) { disp.textContent = fmt(v); disp.style.color = '#1a1a18'; ncEl.textContent = 'NC por devolución: ' + fmt(nc); ncEl.style.color = '#c0392b'; }
else { disp.textContent = fmt(v); disp.style.color = '#1a7a3c'; ncEl.textContent = 'Sin devolución'; ncEl.style.color = '#aaa'; }
}
function nkConfirmar() {
if (!npComp) return;
const v = parseInt(npValor) || 0;
if (v > Math.ceil(npMontoOrig)) { alert('El monto no puede superar el original.'); return; }
if (v <= 0) { alert('Ingresá un monto mayor a cero.'); return; }
montosReales[npComp] = v;

async function guardarProgreso() {
const btn = document.getElementById('btn-progreso');
btn.textContent = '💾 Guardando...';
btn.disabled = true;

const payload = {
action: 'guardarProgreso',
fecha: fechaDia,
codigo: repActual,
estados: estados,
montosReales: montosReales,
montosMixtos: montosMixtos,
motivosBaja: motivosBaja,
};

try {
await fetch(SCRIPT_URL, { method: 'POST', body: JSON.stringify(payload) });
btn.textContent = '✓ Guardado';
setTimeout(() => { btn.textContent = '💾 Guardar'; btn.disabled = false; }, 2000);
} catch(err) {
btn.textContent = '⚠️ Error';
setTimeout(() => { btn.textContent = '💾 Guardar'; btn.disabled = false; }, 2000);
}
}

async function restaurarProgreso(codigo, fecha) {
try {
const resp = await fetch(SCRIPT_URL + `?action=getProgreso&codigo=${codigo}&fecha=${encodeURIComponent(fecha)}`, { method: 'GET' });
const data = await resp.json();
if (data.ok && data.progreso) {
estados = data.progreso.estados || {};
montosReales = data.progreso.montosReales || {};
montosMixtos = data.progreso.montosMixtos || {};
motivosBaja = data.progreso.motivosBaja || {};
return true;
}
} catch(err) {}
return false;
}
